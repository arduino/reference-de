= read

// ÜBERSICHTSABSCHNITT STARTET
[#overview]
--

[float]
=== Beschreibung
Diese Funktion liest ein Byte, das nach einem Aufruf von `requestFrom()` von einem Peripheriegerät an einen Controller übertragen wurde oder das von einem Controller an ein Peripheriegerät übertragen wurde. `read()` erbt von der Stream-Dienstklasse.

[float]
=== Syntax
`Wire.read()`

[float]
=== Parameter

Keine.

[float]
=== Rückgabewert

Das nächste empfangene Byte.

[float]
=== Beispielcode

[source,arduino]
----
#include <Wire.h>

void setup() {
  Wire.begin();             // I2C-Bus beitreten (Adresse ist für den Controller optional)
  Serial.begin(9600);       // Serielle Schnittstelle für die Ausgabe starten
}

void loop() {
    Wire.requestFrom(2, 6);    // 6 Bytes vom Peripheriegerät Nummer 2 anfordern

    // Das Peripheriegerät sendet eventuell weniger als angefordert
    while(Wire.available()) {
        char c = Wire.read();    // Ein Byte als Zeichen empfangen
        Serial.print(c);         // Das Zeichen ausgeben
    }

    delay(500);
}
----
